```python
from pathlib import Path

class EdgeCalibrator:
    """raw_opportunity → expected_return_5d 변환."""

    def __init__(
        self,
        calibration_path: str | Path = "v3/config/edge_calibration.json",
        min_bucket_n: int = 30,
        insufficient_action: Literal["block", "global", "demote"] = "block",
    ):
        """
        Args:
            calibration_path: calibrate_edge.py 산출 JSON.
            min_bucket_n: 분류 가능한 최소 bucket sample.
            insufficient_action:
                - block: reject_reason="insufficient_calibration"
                - global: global average fallback
                - demote: net_edge × 0.5 (보수적)
        """
        self.table: dict[tuple, CalibrationBucket] = self._load(calibration_path)
        self.min_bucket_n = min_bucket_n
        self.insufficient_action = insufficient_action

    def calibrate(
        self,
        candidate: EdgeCandidate,
    ) -> EdgeCandidate:
        """단일 후보에 calibration 적용. Pure function."""
        keys = self._fallback_keys(candidate)
        for key in keys:
            bucket = self.table.get(key)
            if bucket and bucket.n >= self.min_bucket_n:
                return candidate.with_calibration(
                    expected_return_5d=bucket.mean_forward_return_5d,
                    expected_mae_5d=bucket.mean_mae_5d,
                    expected_mfe_5d=bucket.mean_mfe_5d,
                    win_probability_5d=bucket.win_rate_5d,
                    calibration_n=bucket.n,
                    calibration_key=key,
                )
        return self._handle_insufficient(candidate)

    def calibrate_batch(
        self,
        candidates: list[EdgeCandidate],
    ) -> list[EdgeCandidate]:
        """여러 후보 batch 처리."""
        return [self.calibrate(c) for c in candidates]

    def _fallback_keys(self, c: EdgeCandidate) -> list[tuple]:
        """Fallback 순서. Level 4 (sector) 사용 안 함."""
        decile = self._opportunity_decile(c.raw_opportunity)
        return [
            (decile, c.regime, c.liquidity_bucket),
            (decile, c.regime),
            (decile,),
            ("global",),
        ]
```

```python
# v3/strategy/edge_engine.py (신규)
@dataclass(frozen=True)
class CostModel:
    commission_roundtrip: float = 0.0002
    spread_estimate: float = 0.0003
    market_impact_base: float = 0.0005
    slippage_vol_multiplier: float = 0.10
    min_slippage_buffer: float = 0.0010
    max_slippage_buffer: float = 0.0040

    def expected_cost(self, ticker: str) -> float:
        return (
            self.commission_roundtrip
            + self.spread_estimate
            + self.market_impact_base
        )

    def slippage_buffer(self, ticker: str, ticker_vol: float) -> float:
        buf = self.min_slippage_buffer + ticker_vol * self.slippage_vol_multiplier
        return min(max(buf, self.min_slippage_buffer), self.max_slippage_buffer)


@dataclass(frozen=True)
class EdgePolicy:
    horizon_days: int = 5
    entry_threshold: float = 0.0040       # calibration에서 도출 (분위수)
    high_conviction_threshold: float = 0.0090
    lambda_mae: float = 0.20
    calibration_n_demote: int = 100        # n < 100 → net_edge × 0.5


class EdgeEngine:
    """expected_return_5d → net_edge_5d. Cost + risk 차감."""

    def __init__(self, cost_model: CostModel, policy: EdgePolicy):
        self.cost_model = cost_model
        self.policy = policy

    def compute(
        self,
        candidate: EdgeCandidate,
        ticker_vol: float,
    ) -> EdgeCandidate:
        """net_edge_5d 산출. candidate는 calibration 완료 상태여야."""
        if candidate.expected_return_5d is None:
            return candidate.with_calibration(
                entry_pass=False,
                reject_reason="not_calibrated",
            )

        cost = self.cost_model.expected_cost(candidate.ticker)
        slip = self.cost_model.slippage_buffer(candidate.ticker, ticker_vol)
        downside = self.policy.lambda_mae * abs(candidate.expected_mae_5d)

        net_edge = candidate.expected_return_5d - cost - slip - downside

        # Calibration 신뢰도 감쇠
        if candidate.calibration_n is not None and candidate.calibration_n < self.policy.calibration_n_demote:
            net_edge *= 0.5

        passes = net_edge > self.policy.entry_threshold

        return candidate.with_calibration(
            expected_cost=cost,
            slippage_buffer=slip,
            downside_risk_penalty=downside,
            net_edge_5d=net_edge,
            entry_pass=passes,
            reject_reason=None if passes else "net_edge_below_threshold",
        )
```

```python
# v3/strategy/allocation.py (신규)
@dataclass(frozen=True)
class RegimeBudget:
    max_gross_exposure: float       # ≤ 1.00 (leverage 거부)
    target_annual_vol: float


@dataclass(frozen=True)
class PositionConstraints:
    max_positions: int = 3
    max_single_weight: float = 0.40
    min_position_weight: float = 0.15
    max_sector_weight: float = 0.60
    volume_participation_max: float = 0.05
    min_order_amount_krw: float = 5_000_000


class AllocationEngine:
    """net_edge / |expected_mae| 기반 sizing."""

    def __init__(
        self,
        constraints: PositionConstraints,
        regime_budgets: dict[str, RegimeBudget],
        sizer: "VolTargetSizer",   # 기존 VolTargetSizer 통합 (대체 아님)
    ):
        self.constraints = constraints
        self.regime_budgets = regime_budgets
        self.sizer = sizer

    def allocate(
        self,
        candidates: list[EdgeCandidate],
        regime: "Regime",
        existing_positions: list[PositionState],
        correlation_matrix: Optional[np.ndarray] = None,
    ) -> dict[str, float]:
        """
        Returns: {ticker: target_weight}
        Σ target_weight ≤ regime.max_gross_exposure ≤ 1.00
        """
        passing = [c for c in candidates if c.entry_pass and c.net_edge_5d > 0]
        passing.sort(key=lambda c: c.net_edge_5d, reverse=True)
        passing = passing[: self.constraints.max_positions]

        if not passing:
            return {}

        # net_edge / |expected_mae| ratio
        ratios = {}
        for c in passing:
            downside = max(abs(c.expected_mae_5d), 0.005)
            ratios[c.ticker] = c.net_edge_5d / downside

        total = sum(ratios.values())
        raw = {t: r / total for t, r in ratios.items()}

        # regime exposure budget cap
        budget = self.regime_budgets[regime.name]
        max_exposure = min(budget.max_gross_exposure, 1.00)  # 강제 cap

        # Apply VolTargetSizer drag/Kelly on top
        scaled = self._apply_sizer_overlay(raw, passing, regime, correlation_matrix)

        # Constraints
        final = self._apply_constraints(scaled, max_exposure, passing)
        return final
```

```python
# v3/strategy/exit_thesis.py (신규)
@dataclass(frozen=True)
class ExitPolicy:
    hold_min_residual_edge: float = 0.0
    reduce_zone_edge: float = 0.0015
    rotation_margin: float = 0.0025
    thesis_break_direction_threshold: float = 0.0
    max_signal_staleness_hours: float = 16.0
    refresh_at_session_start: bool = True


class ExitThesisEngine:
    """Conditional Veto 정상화 + HOLD/REDUCE/ROTATE/EXIT 의사결정."""

    RISK_BASED_EXITS = {
        "portfolio_stop",
        "dynamic_stop_mae",
        "vol_contraction",
        "circuit_breaker",
    }

    def __init__(
        self,
        edge_engine: EdgeEngine,
        edge_calibrator: EdgeCalibrator,
        policy: ExitPolicy,
        signal_refresh_fn: Optional[callable] = None,
    ):
        self.edge_engine = edge_engine
        self.edge_calibrator = edge_calibrator
        self.policy = policy
        self.signal_refresh_fn = signal_refresh_fn

    def decide(
        self,
        position: PositionState,
        exit_trigger: str,
        current_edge: Optional[EdgeCandidate],
        replacement_candidates: list[EdgeCandidate],
        signal_age_hours: float,
    ) -> BookAction:
        """단일 포지션 exit 결정."""

        # Risk-based: veto 금지
        if exit_trigger in self.RISK_BASED_EXITS:
            return BookAction(
                date=pd.Timestamp.now(),
                action_type="EXIT",
                ticker=position.ticker,
                target_weight=0.0,
                current_weight=position.current_weight,
                reason=f"risk_exit:{exit_trigger}",
                source_policy="ExitThesisEngine",
            )

        # Stale signal refresh
        if signal_age_hours > self.policy.max_signal_staleness_hours:
            if self.signal_refresh_fn:
                current_edge = self.signal_refresh_fn(position.ticker)

        if current_edge is None or current_edge.net_edge_5d is None:
            # Cannot make informed decision → conservative exit
            return BookAction(..., action_type="EXIT", reason="no_fresh_signal")

        # Residual edge negative → exit
        if current_edge.net_edge_5d <= self.policy.hold_min_residual_edge:
            return BookAction(..., action_type="EXIT", reason="residual_edge_negative")

        # Replacement check (Rotation)
        best = self._best_replacement(replacement_candidates, current_edge)
        if best:
            switching_cost = self._switching_cost(position.ticker, best.ticker)
            if best.net_edge_5d - current_edge.net_edge_5d > switching_cost + self.policy.rotation_margin:
                return BookAction(
                    ...,
                    action_type="ROTATE",
                    replacement_ticker=best.ticker,
                    reason=f"rotation_edge={best.net_edge_5d - current_edge.net_edge_5d:.4f}",
                )

        # Edge weakening → reduce
        if current_edge.net_edge_5d < self.policy.reduce_zone_edge:
            return BookAction(
                ...,
                action_type="TRIM",
                target_weight=position.current_weight * 0.7,
                reason="edge_weakened",
            )

        # Hold
        return BookAction(..., action_type="KEEP", reason="thesis_alive")
```

```python
@dataclass(frozen=True)
class TierThresholds:
    """Calibration panel 분위수에서 도출된 임계값."""
    s_tier_net_edge: float          # p90
    a_tier_net_edge: float          # p70
    b_tier_net_edge: float          # p50
    s_tier_winprob: float           # p70 within top decile
    a_tier_winprob: float           # 0.55
    s_tier_max_mae: float           # p30 (less negative = better)
    min_stability_score: dict[str, float] = field(default_factory=lambda: {
        "S": 0.70, "A": 0.50,
    })


class EdgeTierSystem(DecisionPolicy):
    name = "edge_tier"

    def __init__(self, thresholds: TierThresholds, enabled: bool = True):
        self.thresholds = thresholds
        self._enabled = enabled

    @property
    def enabled(self) -> bool:
        return self._enabled

    def assign(self, candidate: EdgeCandidate) -> EdgeCandidate:
        """단일 후보에 tier 할당."""
        tier = self._compute_tier(candidate)
        return candidate.with_calibration(edge_tier=tier)

    def _compute_tier(self, c: EdgeCandidate) -> EdgeTier:
        if c.net_edge_5d is None:
            return "BLOCKED"

        # S-tier
        if (c.net_edge_5d >= self.thresholds.s_tier_net_edge
            and c.win_probability_5d >= self.thresholds.s_tier_winprob
            and c.expected_mae_5d >= self.thresholds.s_tier_max_mae):
            return "S"

        # A-tier
        if (c.net_edge_5d >= self.thresholds.a_tier_net_edge
            and c.win_probability_5d >= self.thresholds.a_tier_winprob):
            return "A"

        # B-tier
        if c.net_edge_5d >= self.thresholds.b_tier_net_edge:
            return "B"

        return "C"
```

```python
@dataclass(frozen=True)
class RotationPolicy:
    enabled: bool = True
    rotation_margin: float = 0.0025
    min_candidate_tier: str = "A"
    allow_s_replace_a: bool = True
    forbid_lower_tier_replace_higher: bool = True
    max_rotations_per_month: int = 4
    min_holding_days_before_rotation: int = 1


class CapitalRotationEngine(DecisionPolicy):
    name = "rotation"

    def __init__(self, policy: RotationPolicy, switching_cost_fn: callable):
        self.policy = policy
        self._switching_cost = switching_cost_fn

    def evaluate(
        self,
        position: PositionState,
        candidate: EdgeCandidate,
        rotations_this_month: int,
    ) -> Optional[BookAction]:
        """rotation 후보 평가. None = no rotation."""
        if not self.policy.enabled:
            return None
        if rotations_this_month >= self.policy.max_rotations_per_month:
            return None
        if position.holding_days < self.policy.min_holding_days_before_rotation:
            return None
        if self._tier_rank(candidate.edge_tier) < self._tier_rank(position.current_tier):
            return None

        rotation_edge = candidate.net_edge_5d - position.residual_edge
        switching_cost = self._switching_cost(position.ticker, candidate.ticker)
        hurdle = switching_cost + self.policy.rotation_margin

        if rotation_edge > hurdle:
            return BookAction(
                date=candidate.date,
                action_type="ROTATE",
                ticker=position.ticker,
                replacement_ticker=candidate.ticker,
                target_weight=0.0,
                current_weight=position.current_weight,
                reason=f"rotation_edge={rotation_edge:.4f} > hurdle={hurdle:.4f}",
                source_policy="CapitalRotationEngine",
                expected_impact=rotation_edge,
            )
        return None
```

```python
# v3/strategy/pyramid.py (신규)
@dataclass(frozen=True)
class PyramidPolicy:
    enabled: bool = True
    add_only_to_winners: bool = True
    forbid_averaging_down: bool = True
    min_unrealized_pnl_for_add: float = 0.015
    min_residual_edge_for_add: float = 0.004
    min_current_tier_for_add: str = "A"
    max_adds_per_position: int = 2
    add_size_fraction_of_initial: float = 0.50
    max_single_weight_after_add: float = 0.40


class PyramidPolicyEngine(DecisionPolicy):
    name = "pyramid"

    def __init__(self, policy: PyramidPolicy):
        self.policy = policy

    def evaluate(
        self,
        position: PositionState,
        candidate: EdgeCandidate,
        adds_so_far: int,
    ) -> Optional[BookAction]:
        if not self.policy.enabled:
            return None
        if adds_so_far >= self.policy.max_adds_per_position:
            return None

        # 절대 금지: 손실 포지션
        if self.policy.forbid_averaging_down and position.pnl_pct <= 0:
            return None

        if position.pnl_pct < self.policy.min_unrealized_pnl_for_add:
            return None

        if candidate.net_edge_5d < self.policy.min_residual_edge_for_add:
            return None

        if self._tier_rank(candidate.edge_tier) < self._tier_rank(self.policy.min_current_tier_for_add):
            return None

        initial = self._initial_weight(position)
        new_weight = min(
            position.current_weight + initial * self.policy.add_size_fraction_of_initial,
            self.policy.max_single_weight_after_add,
        )

        return BookAction(
            date=candidate.date,
            action_type="ADD_TO_WINNER",
            ticker=position.ticker,
            target_weight=new_weight,
            current_weight=position.current_weight,
            reason=(
                f"winner add-on: pnl={position.pnl_pct:.3f}, "
                f"residual_edge={candidate.net_edge_5d:.4f}, "
                f"tier={candidate.edge_tier}"
            ),
            source_policy="PyramidPolicy",
        )
```

### 4.6 Partial Exit

```python
# v3/strategy/partial_exit.py (신규)
@dataclass(frozen=True)
class PartialExitPolicy:
    enabled: bool = True
    trim_fraction_default: float = 0.50
    min_remaining_weight: float = 0.15      # Phase 25.2 sizer floor와 동일
    profit_protection_trigger: float = 0.030
    drawdown_from_peak_trigger: float = -0.015
    trim_if_residual_edge_below: float = 0.002
    exit_if_residual_edge_below: float = 0.000
    full_exit_on_vol_contraction: bool = True


class PartialExitEngine(DecisionPolicy):
    name = "partial_exit"

    def __init__(self, policy: PartialExitPolicy):
        self.policy = policy

    def evaluate(
        self,
        position: PositionState,
        residual_edge: float,
        thesis_break: Optional[str] = None,
    ) -> BookAction:
        # vol contraction → 전량
        if self.policy.full_exit_on_vol_contraction and thesis_break == "vol_contraction":
            return self._exit_full(position, "vol_contraction")

        # residual edge ≤ 0 → 전량
        if residual_edge <= self.policy.exit_if_residual_edge_below:
            return self._exit_full(position, "residual_edge_negative")

        # Profit protection: +3% 이후 -1.5% 하락 + edge 약화
        if (position.max_unrealized_pnl_pct >= self.policy.profit_protection_trigger
            and position.drawdown_from_peak_pct <= self.policy.drawdown_from_peak_trigger
            and residual_edge < self.policy.trim_if_residual_edge_below):
            return self._trim(position, "winner_protection_trim")

        # Edge weakening → 30% trim
        if residual_edge < self.policy.trim_if_residual_edge_below:
            return self._trim(position, "weak_residual_edge", fraction=0.30)

        return BookAction(..., action_type="KEEP")

    def _trim(self, position, reason, fraction=None) -> BookAction:
        f = fraction or self.policy.trim_fraction_default
        new_weight = position.current_weight * (1 - f)
        # 잔여가 floor 미만이면 전량
        if new_weight < self.policy.min_remaining_weight:
            return self._exit_full(position, f"{reason}_below_floor")
        return BookAction(
            ...,
            action_type="TRIM",
            target_weight=new_weight,
            reason=reason,
        )
```

```python
@dataclass(frozen=True)
class TCSnapshot:
    date: pd.Timestamp
    transfer_coefficient: float
    top_decile_capture: float
    n_candidates: int
    n_deployed: int
    total_deployed_weight: float


class TransferCoefficientMonitor(DiagnosticSink):
    name = "tc_monitor"

    def __init__(
        self,
        log_path: str | Path = "v3/saved_models/tc_history.jsonl",
        warning_threshold: float = 0.30,
        critical_threshold: float = 0.10,
    ):
        self.log_path = Path(log_path)
        self.warning_threshold = warning_threshold
        self.critical_threshold = critical_threshold
        self._enabled = True

    def compute(
        self,
        candidates: list[EdgeCandidate],
        final_weights: dict[str, float],
    ) -> TCSnapshot:
        from scipy.stats import spearmanr

        edges = [c.net_edge_5d or 0.0 for c in candidates]
        weights = [final_weights.get(c.ticker, 0.0) for c in candidates]
        tc, _ = spearmanr(edges, weights) if len(edges) > 1 else (0.0, None)

        # Top decile capture
        sorted_cands = sorted(candidates, key=lambda c: c.net_edge_5d or 0.0, reverse=True)
        top_n = max(1, len(sorted_cands) // 10)
        top_tickers = {c.ticker for c in sorted_cands[:top_n]}
        top_weight = sum(w for t, w in final_weights.items() if t in top_tickers)
        total = sum(final_weights.values())
        capture = top_weight / total if total > 0 else 0.0

        return TCSnapshot(
            date=pd.Timestamp.now(),
            transfer_coefficient=tc,
            top_decile_capture=capture,
            n_candidates=len(candidates),
            n_deployed=len([w for w in final_weights.values() if w > 0]),
            total_deployed_weight=total,
        )
```

```python
# v3/strategy/book_optimizer.py (신규, V3.3 핵심)
@dataclass(frozen=True)
class FeatureFlags:

    @classmethod

        """V3.2.1 동작 모드 (회귀 테스트용)."""
        return cls(no_trade_logger=False, tc_monitor=False, execution_quality=False)


class BookOptimizer:
    """
    V3.3 의사결정 엔진. SignalGenerator를 wrap하고 Edge layer 추가.
    backtest engine과 live pipeline에서 동일하게 호출.
    """

    def __init__(
        self,
        signal_gen: "SignalGenerator",
        edge_calibrator: EdgeCalibrator,
        edge_engine: EdgeEngine,
        edge_tier: EdgeTierSystem,
        allocation: AllocationEngine,
        exit_thesis: ExitThesisEngine,
        rotation: CapitalRotationEngine,
        pyramid: PyramidPolicyEngine,
        signal_decay: SignalDecayEngine,
        partial_exit: PartialExitEngine,
        no_trade_logger: NoTradeReasonLogger,
        tc_monitor: TransferCoefficientMonitor,
        execution_quality: ExecutionQualityMonitor,
        features: FeatureFlags,
    ):
        self.signal_gen = signal_gen
        self.edge_calibrator = edge_calibrator
        # ... (모든 정책 모듈 의존성 주입)
        self.features = features

    def decide(
        self,
        ohlcv: pd.DataFrame,
        vol_scores: pd.DataFrame,
        regime: "Regime",
        cost: float,
        state: "OperationalState",
        positions: list[PositionState],
        ticker_volume_map: dict[str, float] = None,
        sector_map: dict[str, str] = None,
    ) -> list[BookAction]:
        """일일 의사결정. backtest/live 공통."""

        # Step 1: Existing SignalGenerator
        signal = self.signal_gen.generate(
            ohlcv=ohlcv,
            vol_scores=vol_scores,
            regime=regime,
            cost=cost,
            state=state,
            ticker_volume_map=ticker_volume_map,
            sector_map=sector_map,
        )

        # Build EdgeCandidate list from signal (always run)
        candidates = self._build_candidates(signal, regime)

        # Step 2: Edge layer (feature-flagged)
        if self.features.edge_calibrator:
            candidates = [self.edge_calibrator.calibrate(c) for c in candidates]
        if self.features.edge_engine:
            candidates = [self.edge_engine.compute(c, ticker_vol=...) for c in candidates]
        if self.features.edge_tier:
            candidates = [self.edge_tier.assign(c) for c in candidates]

        actions: list[BookAction] = []

        # Step 3: Position-level decisions
        for position in positions:
            current_edge = self._find_candidate(candidates, position.ticker)

            # Exit decision
            if self.features.exit_thesis:
                action = self.exit_thesis.decide(
                    position=position,
                    exit_trigger=...,
                    current_edge=current_edge,
                    replacement_candidates=candidates,
                    signal_age_hours=...,
                )
            else:
                action = self._legacy_exit(position)
            actions.append(action)

            # Pyramid add-on (only if KEEP)
            if action.action_type == "KEEP" and self.features.pyramid:
                add = self.pyramid.evaluate(position, current_edge, adds_so_far=...)
                if add:
                    actions.append(add)

        # Step 4: New entries
        if self.features.allocation:
            new_weights = self.allocation.allocate(
                candidates=candidates,
                regime=regime,
                existing_positions=positions,
            )
        else:
            new_weights = self._legacy_allocate(signal)

        for ticker, weight in new_weights.items():
            actions.append(BookAction(
                date=...,
                action_type="ADD_NEW",
                ticker=ticker,
                target_weight=weight,
                current_weight=0.0,
                reason="new entry",
                source_policy="AllocationEngine" if self.features.allocation else "VolTargetSizer",
            ))

        # Step 5: Diagnostics (always read-only)
        if self.features.no_trade_logger:
            self._record_no_trades(candidates, actions)
        if self.features.tc_monitor:
            tc = self.tc_monitor.compute(candidates, new_weights)
            self.tc_monitor.record({"snapshot": tc})

        # Final: action priority sort
        actions = self._prioritize(actions)
        return actions

    def _prioritize(self, actions: list[BookAction]) -> list[BookAction]:
        """Action 우선순위 정렬."""
        priority = {
            "EXIT": 0,           # risk exit first
            "BLOCKED": 1,
            "ROTATE": 2,
            "ADD_TO_WINNER": 3,  # pyramid before new entry
            "ADD_NEW": 4,
            "TRIM": 5,
            "KEEP": 6,
            "NO_ACTION": 7,
        }
        return sorted(actions, key=lambda a: priority.get(a.action_type, 99))
```
